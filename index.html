import pandas as pd from PIL import Image, ImageDraw, ImageFont import os # Create output folder if it doesn't exist output_dir = "render_PNG" if not os.path.exists(output_dir): os.makedirs(output_dir) # Read Excel File df = pd.read_excel("data.xlsx") # Load Font font_path = "BAHI.ttf" font_size = 50 font = ImageFont.truetype(font_path, font_size) # Load Logo logo_path = "IMK_SPORT_2.png" logo = Image.open(logo_path).convert("RGBA") # Frame size setup canvas_width = 1920 canvas_height = 100 for index, row in df.iterrows(): text = str(row['ticker']) # Create transparent canvas base_img = Image.new("RGBA", (canvas_width, canvas_height), (0, 0, 0, 0)) draw = ImageDraw.Draw(base_img) # Calculate Text Position (Y-axis centered) bbox = font.getbbox(text) text_height = bbox[3] - bbox[1] text_y = (canvas_height - text_height) // 2 # Draw text in WHITE color draw.text((10, text_y), text, font=font, fill=(255, 255, 255, 255)) # Position and paste Logo (Right side) logo_x = canvas_width - logo.width - 10 logo_y = (canvas_height - logo.height) // 2 # Paste logo preserving transparency mask base_img.paste(logo, (logo_x, logo_y), logo) # Save output frame output_path = os.path.join(output_dir, f"frame_{index:04d}.png") base_img.save(output_path, "PNG") print("ڕەندەرکردن بە سەرکەوتوویی تەواو بوو!")
